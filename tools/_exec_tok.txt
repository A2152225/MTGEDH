function normalizeTokenTypeText(value: unknown): string {
  return String(value || '')
    .replace(/\u2019/g, "'")
    .toLowerCase()
    .replace(/\b(?:token|tokens|artifact|creature|enchantment)\b/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}
