190|  if (!selector) return '';

192|    kind: 'battlefield_selector';
193|    types?: readonly string[];
194|    controllerFilter?: string;

196|    if (!value || typeof value !== 'object') return false;

200|  if (legacyBattlefieldSelector(runtimeSelector)) {
201|    const types = Array.isArray(runtimeSelector.types) ? runtimeSelector.types.join(' ') : '';
202|    const controller = runtimeSelector.controllerFilter ? `${runtimeSelector.controllerFilter} ` : '';
203|    return `${controller}${types}`.trim();
204|  }
205|  if (selector.kind === 'raw') return String(selector.text || '').trim();
206|  return String((selector as { text?: string; raw?: string }).text || (selector as { raw?: string }).raw || '').trim();
